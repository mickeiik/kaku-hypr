# Fedora packages kaku-hypr needs. Installed by install.sh, which first adds the
# lionheartp/Hyprland COPR (Hyprland isn't in Fedora's repos), and can also add voxtype,
# for speech to text, from its GitHub releases.
hyprland                # the compositor (COPR)
xdg-desktop-portal-hyprland  # screen sharing, screenshots for apps (COPR)
xdg-desktop-portal-gtk  # file pickers
sway                    # runs the login screen (SDDM's compositor on Fedora)
sway-config-fedora      # the Sway session, kept as a fallback at the login screen
swayidle                # lock and screen off when idle (works on Hyprland)
swaylock                # fallback lock screen if Quickshell isn't running
sddm                    # login manager (kaku login theme)
quickshell              # the whole shell: bar, panels, launcher, notifications, lock screen...
foot                    # terminal
Thunar                  # file manager
micro                   # text editor
adw-gtk3-theme          # GTK3 apps in libadwaita style
papirus-icon-theme      # icons
jetbrains-mono-fonts-all    # font
fontawesome-6-free-fonts    # bar and panel icons
brightnessctl           # screen brightness (laptop screen)
ddcutil                 # screen brightness (monitors, DDC/CI)
playerctl               # media players (Now Playing)
wlsunset                # night light
wl-clipboard            # clipboard
cliphist                # clipboard history (Ctrl+Super+V)
wtype                   # dictation typing
grim                    # screenshots
slurp                   # screenshot area selection
swappy                  # screenshot annotation
wf-recorder             # screen recording
libnotify               # notify-send
jq                      # used by scripts
xdg-utils               # xdg-mime: the ntfy panel opens attachments in their app
fzf                     # package browsers of the launcher (Install / Remove)
fprintd                 # fingerprint (lock screen, admin prompt)
fprintd-pam
oo7-daemon              # keyring that keeps app sign-ins (Claude Desktop, VS Code...), unlocked by the login
pam_oo7
dnscrypt-proxy          # system-wide DNS over HTTPS (Mullvad)
